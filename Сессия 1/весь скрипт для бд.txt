AS
BEGIN
    SET NOCOUNT ON;
    
    SELECT 
        пп.ID,
        пп.НомерПартии,
        прод.Наименование AS Продукт,

        шт.НаименованиеШага AS ТекущийШаг,
        CASE 
            WHEN EXISTS (SELECT 1 FROM [dbo].[ВыполнениеШаговПартии] WHERE ПартияПроизводстваID = пп.ID AND Отклонение = 1)
            THEN 1
            ELSE 0
        END AS ЕстьОтклонения
    FROM [dbo].[ПроизводственныеПартии] пп
    INNER JOIN [dbo].[ПроизводственныеЗаказы] пз ON пп.ЗаказID = пз.ID
    INNER JOIN [dbo].[Продукция] прод ON пз.ПродуктID = прод.ID
    LEFT JOIN [dbo].[ШагиТехКарты] шт ON пп.ТекущийШагID = шт.ID
    WHERE пп.Статус IN ('выполняется', 'запланирована')
    ORDER BY пп.ВремяСтарта;
END;

-- =====================================================
-- ПРОЦЕДУРА 9: Отчет по использованию рецептур
-- =====================================================
CREATE   PROCEDURE [dbo].[Отчет_ИспользованиеРецептур]
    @ДатаНачала DATE,
    @ДатаОкончания DATE
AS
BEGIN
    SET NOCOUNT ON;
    
    SELECT 
        р.ID AS РецептID,
        прод.Наименование AS Продукт,
        р.Версия,
        COUNT(пп.ID) AS КоличествоПартий,
        SUM(ISNULL(пп.ФактКоличество_кг, 0)) AS ОбъемПроизводства_кг,
        AVG(ISNULL(пп.ФактКоличество_кг, 0)) AS СреднийОбъем_кг
    FROM [dbo].[Рецептуры] р
    INNER JOIN [dbo].[Продукция] прод ON р.ПродуктID = прод.ID
    LEFT JOIN [dbo].[ПроизводственныеЗаказы] пз ON р.ID = пз.РецептID
    LEFT JOIN [dbo].[ПроизводственныеПартии] пп ON пз.ID = пп.ЗаказID
    WHERE (пп.ВремяСтарта IS NULL OR CAST(пп.ВремяСтарта AS DATE) BETWEEN @ДатаНачала AND @ДатаОкончания)
    GROUP BY р.ID, прод.Наименование, р.Версия
    ORDER BY ОбъемПроизводства_кг DESC;
END;
GO

-- ПРОЦЕДУРА 10: Отчет по лабораторным блокировкам
-- =====================================================
CREATE   PROCEDURE [dbo].[Отчет_ЛабораторныеБлокировки]
    @ДатаНачала DATE,
    @ДатаОкончания DATE
AS
BEGIN
    SET NOCOUNT ON;
    
    SELECT 
        ли.ID AS ИспытаниеID,
        ISNULL(пп.НомерПартии, пс.НомерПартии) AS НомерПартии,
        ISNULL(прод.Наименование, сырье.Наименование) AS Наименование,
        ли.ДатаАнализа AS ДатаБлокировки,
        ли.Решение,
        ли.КомментарийАналитика AS Причина,
        пол.ПолноеИмя AS Ответственный
    FROM [dbo].[ЛабораторныеИспытания] ли
    LEFT JOIN [dbo].[ПроизводственныеПартии] пп ON ли.ПартияПроизводстваID = пп.ID
    LEFT JOIN [dbo].[ПартииСырья] пс ON ли.ПартияСырьяID = пс.ID
    LEFT JOIN [dbo].[ПроизводственныеЗаказы] пз ON пп.ЗаказID = пз.ID
    LEFT JOIN [dbo].[Продукция] прод ON пз.ПродуктID = прод.ID
    LEFT JOIN [dbo].[Сырье] ON пс.СырьеID = [dbo].[Сырье].ID
    LEFT JOIN [dbo].[Пользователи] пол ON ли.АналитикID = пол.ID
    WHERE ли.Решение = 'blocked'
        AND CAST(ли.ДатаАнализа AS DATE) BETWEEN @ДатаНачала AND @ДатаОкончания
    ORDER BY ли.ДатаАнализа DESC;
END;

-- Отчет по партиям за период
CREATE   PROCEDURE [dbo].[Отчет_ПартииЗаПериод]
    @ДатаНачала DATE,
    @ДатаОкончания DATE
AS
BEGIN
    SELECT 
        пп.НомерПартии,
        прод.Наименование AS Продукт,
        пп.ВремяСтарта AS ДатаСтарта,
        пп.ВремяОкончания AS ДатаОкончания,
        пп.Статус,
        пп.ФактКоличество_кг,
        CASE 
            WHEN EXISTS (SELECT 1 FROM [dbo].[ЛабораторныеИспытания] ли 
                         WHERE ли.ПартияПроизводстваID = пп.ID AND ли.Результат = 'fail')
            THEN 'Есть отклонения'
            ELSE 'Нет отклонений'
        END AS НаличиеОтклонений,
        ISNULL((
            SELECT TOP 1 Решение 
            FROM [dbo].[ЛабораторныеИспытания] 
            WHERE ПартияПроизводстваID = пп.ID 
            ORDER BY ДатаАнализа DESC
        ), 'не проверено') AS ЛабораторноеРешение
    FROM [dbo].[ПроизводственныеПартии] пп
    INNER JOIN [dbo].[ПроизводственныеЗаказы] пз ON пп.ЗаказID = пз.ID
    INNER JOIN [dbo].[Продукция] прод ON пз.ПродуктID = прод.ID
    WHERE CAST(пп.ВремяСтарта AS DATE) BETWEEN @ДатаНачала AND @ДатаОкончания
    ORDER BY пп.ВремяСтарта DESC;
END;

CREATE   PROCEDURE [dbo].[ПартииОжидающиеКонтроля]
AS
BEGIN
    SET NOCOUNT ON;
    
    SELECT 
        пп.ID,
        пп.НомерПартии,
        прод.Наименование AS Продукт,

        CASE

            THEN 'Испытание назначено'
            ELSE 'Ожидает назначения'
        END AS СтатусКонтроля
    FROM [dbo].[ПроизводственныеПартии] пп
    INNER JOIN [dbo].[ПроизводственныеЗаказы] пз ON пп.ЗаказID = пз.ID
    INNER JOIN [dbo].[Продукция] прод ON пз.ПродуктID = прод.ID
    WHERE пп.Статус = 'завершена'
        AND NOT EXISTS (
            SELECT 1 FROM [dbo].[ЛабораторныеИспытания] 
            WHERE ПартияПроизводстваID = пп.ID AND Решение IS NOT NULL
        )
    ORDER BY пп.ВремяОкончания;
END;

-- =====================================================
-- ПРОЦЕДУРА 13: Статистика по отклонениям
-- =====================================================
CREATE   PROCEDURE [dbo].[СтатистикаОтклонений]
    @ДатаНачала DATE,
    @ДатаОкончания DATE
AS
BEGIN
    SET NOCOUNT ON;
    
    SELECT 
        шт.НаименованиеШага,
        COUNT(*) AS КоличествоОтклонений,
        COUNT(DISTINCT пп.НомерПартии) AS КоличествоПартий,
        AVG(ABS(ISNULL(вп.ФактТемпература - шт.ПланТемпература, 0))) AS СреднееОтклонениеТемпературы
    FROM [dbo].[ВыполнениеШаговПартии] вп
    INNER JOIN [dbo].[ПроизводственныеПартии] пп ON вп.ПартияПроизводстваID = пп.ID
    INNER JOIN [dbo].[ШагиТехКарты] шт ON вп.ШагТехКартыID = шт.ID
    WHERE вп.Отклонение = 1
        AND CAST(вп.ВремяСтарта AS DATE) BETWEEN @ДатаНачала AND @ДатаОкончания
    GROUP BY шт.НаименованиеШага
    ORDER BY КоличествоОтклонений DESC;
END;
